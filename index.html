Seamless Integration

npm install ghostsearch flexsearch
# or using yarn
yarn add ghostsearch flexsearch
                    
import { GhostSearch } from 'ghostsearch';

const db = new GhostSearch({
  index: ['title', 'content', 'path'],
  tokenize: 'forward',
  cache: true
});

// Load records
await db.add(myHugeDataset);
                    
// Fires on keyup, returns in <1ms
const results = db.search("dashboard settings", {
  limit: 50,
  suggest: true 
});

console.log(results);
// [{ id: 1, title: 'Dashboard Settings', ... }, ...]
